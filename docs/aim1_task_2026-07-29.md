## Step 5：冻结 CHARLS 系数，应用于 CLHLS

**离散时间模型的4年累积概率公式：**
P(4y death) = 1 − (1−h₁)(1−h₂)

其中 h_t = logistic(β_FI·fi + β_age·age + β_female·female + γ_t)，γ_t 是 period t 的截距。

```r
coef_b <- coef(fit_b)
# 提取：intercept、fi_full、age、female 系数
# period 系数：coef_b["(Intercept)"] 是 period1 的截距
#              coef_b["factor(period)2"] 是 period2 相对于 period1 的增量

beta_intercept <- coef_b["(Intercept)"]
beta_fi        <- coef_b["fi_full"]
beta_age       <- coef_b["age"]
beta_female    <- coef_b["female"]
gamma2_diff    <- coef_b["factor(period)2"]  # period2 relative to period1

# 对 CLHLS 计算线性预测值（不含 period 效应，用期均值）
lp_base <- beta_intercept + beta_fi * clhls_df$fi_full +
           beta_age * clhls_df$age + beta_female * clhls_df$female

# h1 = logistic(lp_base), h2 = logistic(lp_base + gamma2_diff)
h1 <- plogis(lp_base)
h2 <- plogis(lp_base + gamma2_diff)

# 4-year cumulative probability
clhls_df$pred_4y <- 1 - (1 - h1) * (1 - h2)

cat("CLHLS pred_4y summary:\n"); print(summary(clhls_df$pred_4y))
cat("Observed 4y mortality rate:", mean(clhls_df$event_4y), "\n")
cat("Predicted 4y mortality rate:", mean(clhls_df$pred_4y), "\n")
```

## Step 6：CLHLS 外部验证指标

```r
library(pROC); library(dcurves); library(ggplot2)

# 1. Discrimination
roc_cl <- roc(clhls_df$event_4y, clhls_df$pred_4y, quiet=TRUE)
c_index_cl <- as.numeric(auc(roc_cl))
cat("CLHLS C-index:", round(c_index_cl, 3), "\n")

# Bootstrap CI (500次，个体级别)
set.seed(2026)
boot_c <- replicate(500, {
  idx <- sample(nrow(clhls_df), replace=TRUE)
  as.numeric(auc(roc(clhls_df$event_4y[idx], clhls_df$pred_4y[idx], quiet=TRUE)))
})
cat("CLHLS C-index 95% CI:", round(quantile(boot_c, c(0.025, 0.975)), 3), "\n")

# 2. Calibration
oe_ratio   <- mean(clhls_df$event_4y) / mean(clhls_df$pred_4y)
lp_pred    <- qlogis(clhls_df$pred_4y)  # logit of predicted
cal_model  <- glm(event_4y ~ offset(lp_pred), data=clhls_df, family=binomial)
cal_interc <- coef(cal_model)["(Intercept)"]   # intercept (should be ~0)
cal_slope_model <- glm(event_4y ~ lp_pred, data=clhls_df, family=binomial)
cal_slope  <- coef(cal_slope_model)["lp_pred"]  # should be ~1

cat("O:E ratio:", round(oe_ratio, 3), "\n")
cat("Calibration intercept:", round(cal_interc, 3), "\n")
cat("Calibration slope:", round(cal_slope, 3), "\n")

# 3. Brier Score
brier <- mean((clhls_df$event_4y - clhls_df$pred_4y)^2)
# IPA：需要 null model
null_rate <- mean(clhls_df$event_4y)
brier_null <- mean((clhls_df$event_4y - null_rate)^2)
ipa <- 1 - brier / brier_null
cat("Brier Score:", round(brier, 4), "| IPA:", round(ipa, 4), "\n")
```
